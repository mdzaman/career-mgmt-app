AWSTemplateFormatVersion: '2010-09-09'
Description: 'API Management Service Infrastructure'

Parameters:
  Environment:
    Type: String
    Default: prod
    AllowedValues: [dev, staging, prod]

Resources:
  # DynamoDB Tables
  APIsTable:
    Type: AWS::DynamoDB::Table
    Properties:
      TableName: !Sub ${Environment}-apis
      BillingMode: PAY_PER_REQUEST
      AttributeDefinitions:
        - AttributeName: id
          AttributeType: S
      KeySchema:
        - AttributeName: id
          KeyType: HASH
      TimeToLiveSpecification:
        AttributeName: ttl
        Enabled: true

  TokensTable:
    Type: AWS::DynamoDB::Table
    Properties:
      TableName: !Sub ${Environment}-tokens
      BillingMode: PAY_PER_REQUEST
      AttributeDefinitions:
        - AttributeName: id
          AttributeType: S
      KeySchema:
        - AttributeName: id
          KeyType: HASH
      TimeToLiveSpecification:
        AttributeName: ttl
        Enabled: true

  # Lambda Functions
  APIManagementFunction:
    Type: AWS::Serverless::Function
    Properties:
      CodeUri: ./src
      Handler: api_management.handler
      Runtime: python3.9
      Timeout: 30
      MemorySize: 256
      Environment:
        Variables:
          APIS_TABLE: !Ref APIsTable
          TOKENS_TABLE: !Ref TokensTable
      Policies:
        - DynamoDBCrudPolicy:
            TableName: !Ref APIsTable
        - DynamoDBCrudPolicy:
            TableName: !Ref TokensTable
        - CloudWatchMetricsPolicy: {}
      Events:
        ListAPIs:
          Type: Api
          Properties:
            Path: /apis
            Method: GET
        CreateAPI:
          Type: Api
          Properties:
            Path: /apis
            Method: POST
        UpdateAPIConfig:
          Type: Api
          Properties:
            Path: /apis/{id}/config
            Method: PUT
        GenerateToken:
          Type: Api
          Properties:
            Path: /tokens
            Method: POST
        RevokeToken:
          Type: Api
          Properties:
            Path: /tokens/{id}/revoke
            Method: POST
        GetMetrics:
          Type: Api
          Properties:
            Path: /metrics
            Method: GET

  # CloudWatch Dashboard
  APIManagementDashboard:
    Type: AWS::CloudWatch::Dashboard
    Properties:
      DashboardName: !Sub ${Environment}-api-management
      DashboardBody: !Sub |
        {
          "widgets": [
            {
              "type": "metric",
              "properties": {
                "metrics": [
                  ["APIGateway", "Count", "ApiId", "*"]
                ],
                "period": 300,
                "stat": "Sum",
                "region": "${AWS::Region}",
                "title": "API Requests"
              }
            },
            {
              "type": "metric",
              "properties": {
                "metrics": [
                  ["APIGateway", "Latency", "ApiId", "*"]
                ],
                "period": 300,
                "stat": "Average",
                "region": "${AWS::Region}",
                "title": "API Latency"
              }
            }
          ]
        }

  # Alarms
  HighErrorRateAlarm:
    Type: AWS::CloudWatch::Alarm
    Properties:
      AlarmName: !Sub ${Environment}-api-error-rate
      MetricName: 5XXError
      Namespace: AWS/ApiGateway
      Statistic: Sum
      Period: 300
      EvaluationPeriods: 2
      Threshold: 5
      ComparisonOperator: GreaterThanThreshold
      AlarmActions:
        - !Ref AlertsTopic

  # SNS Topic for Alerts
  AlertsTopic:
    Type: AWS::SNS::Topic
    Properties:
      TopicName: !Sub ${Environment}-api-alerts

Outputs:
  APIEndpoint:
    Description: API Gateway endpoint URL
    Value: !Sub https://${ServerlessRestApi}.execute-api.${AWS::Region}.amazonaws.com/${Environment}

  DashboardURL:
    Description: CloudWatch Dashboard URL
    Value: !Sub https://${AWS::Region}.console.aws.amazon.com/cloudwatch/home?region=${AWS::Region}#dashboards:name=${APIManagementDashboard}
